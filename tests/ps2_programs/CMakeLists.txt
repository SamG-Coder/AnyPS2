cmake_minimum_required(VERSION 3.24)
project(AnyPS2OriginalPs2Programs LANGUAGES NONE)
if(NOT PS2DEV)
    message(FATAL_ERROR "Pass -DPS2DEV=<PS2DEV toolchain directory>")
endif()
find_program(ee_compiler mips64r5900el-ps2-elf-g++ HINTS "${PS2DEV}/ee/bin" NO_DEFAULT_PATH REQUIRED)
execute_process(COMMAND "${ee_compiler}" --version RESULT_VARIABLE status OUTPUT_VARIABLE version ERROR_VARIABLE error)
if(NOT status EQUAL 0)
    message(FATAL_ERROR "PS2 cross-compiler cannot start (${status}). On Windows put matching 32-bit MinGW host DLLs on PATH. ${error}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiler-version.txt" "${version}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cases.json" catalog)
set(PS2_TEST_PROFILES "O0;O2" CACHE STRING "Selected PS2 compiler profiles from cases.json")
string(JSON profile_count LENGTH "${catalog}" profiles)
math(EXPR profile_last "${profile_count}-1")
set(known_profiles)
foreach(index RANGE ${profile_last})
    string(JSON profile GET "${catalog}" profiles ${index} name)
    list(APPEND known_profiles "${profile}")
endforeach()
foreach(profile IN LISTS PS2_TEST_PROFILES)
    if(NOT profile IN_LIST known_profiles)
        message(FATAL_ERROR "Unknown PS2 test profile: ${profile}")
    endif()
endforeach()
string(JSON count LENGTH "${catalog}" cases)
math(EXPR last "${count}-1")
foreach(index RANGE ${last})
    string(JSON name GET "${catalog}" cases ${index} name)
    string(JSON case_id GET "${catalog}" cases ${index} id)
    string(JSON source_count LENGTH "${catalog}" cases ${index} sources)
    math(EXPR source_last "${source_count}-1")
    set(sources)
    foreach(source_index RANGE ${source_last})
        string(JSON source GET "${catalog}" cases ${index} sources ${source_index})
        list(APPEND sources "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
    endforeach()
    foreach(profile_index RANGE ${profile_last})
        string(JSON profile GET "${catalog}" profiles ${profile_index} name)
        if(NOT profile IN_LIST PS2_TEST_PROFILES)
            continue()
        endif()
        string(JSON base GET "${catalog}" profiles ${profile_index} base)
        string(JSON flag_count LENGTH "${catalog}" profiles ${profile_index} flags)
        math(EXPR flag_last "${flag_count}-1")
        set(flags)
        foreach(flag_index RANGE ${flag_last})
            string(JSON flag GET "${catalog}" profiles ${profile_index} flags ${flag_index})
            list(APPEND flags "${flag}")
        endforeach()
        set(output "${CMAKE_CURRENT_BINARY_DIR}/${name}-${profile}.elf")
        add_custom_command(OUTPUT "${output}"
            BYPRODUCTS "${output}.map"
            COMMAND "${ee_compiler}" -std=c++20 ${flags} -G0 -mno-abicalls -fno-pic
                -ffreestanding -fno-exceptions -fno-rtti -fno-stack-protector -fno-unwind-tables
                -fno-asynchronous-unwind-tables -nostdlib -Wall -Wextra -Werror
                "-DTEST_CASE_ID=${case_id}" "${CMAKE_CURRENT_SOURCE_DIR}/start.S"
                "${CMAKE_CURRENT_SOURCE_DIR}/runtime.cpp" ${sources}
                "-Wl,--defsym=test_image_base=${base},-T,${CMAKE_CURRENT_SOURCE_DIR}/link.ld,-Map,${output}.map,--build-id=none"
                -o "${output}"
            DEPENDS start.S runtime.cpp protocol.hpp callback_helpers.hpp object_helpers.hpp
                ${sources} link.ld cases.json "${ee_compiler}"
            VERBATIM)
        add_custom_target(${name}-${profile} ALL DEPENDS "${output}")
    endforeach()
endforeach()
